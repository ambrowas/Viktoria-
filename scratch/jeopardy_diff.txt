+                <span className="text-[10px] font-black uppercase tracking-[0.2em] text-[#fca311]">
+                  Panel of Elders - Clue Inspection
+                </span>
+                <h2 className="text-2xl md:text-3xl font-black text-white uppercase tracking-tight">
+                  {active.category.name} · <span className="text-[#fca311]">{active.question.points} points</span>
+                </h2>
+              </div>
+              <button
+                onClick={clearQuestionState}
+                className="p-3 rounded-2xl bg-white/5 hover:bg-red-500/10 hover:text-red-400 text-slate-400 border border-white/5 transition-all active:scale-95 flex items-center justify-center"

+
+            <div className="flex-1 min-h-0 overflow-y-auto grid grid-cols-1 md:grid-cols-2 gap-6 md:gap-8 pr-2">

+              <div className="space-y-6 flex flex-col justify-start">
+                <div className="space-y-2">
+                  <span className="text-[10px] font-black uppercase tracking-widest text-[#fca311]/80">
+                    The Question / Clue
+                  </span>

+                  </div>

+                <div className="space-y-2">
+                  <span className="text-[10px] font-black uppercase tracking-widest text-emerald-400">

+                </div>

+                  <span className="text-[10px] font-black uppercase tracking-widest text-[#fca311]/80">
+                    Explanation

+                {active.question.answerMediaUrl && active.question.answerMediaType && (

+                    <span className="text-[10px] font-black uppercase tracking-widest text-slate-400">
+                      Answer Media
+                    </span>
+                    <div className="bg-slate-900/40 border border-white/5 rounded-2xl p-4 flex justify-center w-full">
+                      {active.question.answerMediaType === "IMAGE" && (
+                        <img
+                          src={resolveMediaUrl(active.question.answerMediaUrl)}
+                          alt="Answer media"
+                          className="max-h-60 w-auto rounded-lg shadow-md object-contain"
+                        />
+                      )}
+                      {active.question.answerMediaType === "AUDIO" && (
+                        <audio
+                          controls
+                          src={resolveMediaUrl(active.question.answerMediaUrl)}
+                          className="w-full"
+                        />
+                      )}
+                      {active.question.answerMediaType === "VIDEO" && (
+                        <video
+                          controls
+                          src={resolveMediaUrl(active.question.answerMediaUrl)}
+                          className="max-h-60 w-auto rounded-lg shadow-md"
+                        />
+                      )}
+                    </div>
+                  </div>
+                )}

+                {/* Observation Submission Section */}
+                <div className="space-y-2 border-t border-white/5 pt-4">
+                  <span className="text-[10px] font-black uppercase tracking-widest text-violet-400">
+                    Submit Observation to Organizers
+                  </span>
+                  <div className="space-y-3">
+                    <textarea
+                      value={observationComment}
+                      onChange={(e) => {
+                        setObservationComment(e.target.value);
+                        setObservationSubmittedMsg("");
+                      }}
+                      placeholder="Write a comment, note or error alert for this card..."
+                      className="w-full bg-slate-900/60 border border-white/5 rounded-xl p-3 text-sm text-slate-300 outline-none focus:border-violet-500/40 focus:ring-0 resize-none h-20"
+                    />
+                    <div className="flex items-center justify-between">
+                      <button
+                        onClick={handleSubmitObservation}
+                        disabled={!observationComment.trim()}

+            <footer className="border-t border-white/10 pt-4 flex justify-end shrink-0">

+                onClick={clearQuestionState}
+                className="px-6 py-3 rounded-2xl bg-white/5 hover:bg-white/10 text-white font-extrabold text-xs uppercase tracking-wider transition-all border border-white/5 active:scale-95"

+                Close Verification

       {feedback && <FeedbackOverlay type={feedback} />}
